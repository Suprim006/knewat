"""One-off: engram/recall old (HEAD) vs new (scratch/newq) on fixtures where every rule fires. Prints PASS/diffs."""

from __future__ import annotations

import random
import sys
import uuid

sys.path.insert(0, "C:/Change_folder_after_HDD/wt-recall/plexus")
from plexus.embed import HashEmbedder
from plexus.engram import Engram
from plexus.graph.client import GraphClient
from plexus.graph.queries import QueryLibrary
from plexus.graph.schema import apply_schema

S = "C:/Users/ACER/AppData/Local/hermes/cache/scratch"
old_lib, new_lib = QueryLibrary(f"{S}/oldq"), QueryLibrary(f"{S}/newq")
client = GraphClient(host="127.0.0.1", port=6375, library=old_lib)
graph = f"test_{uuid.uuid4().hex[:12]}"
try:
    apply_schema(client, graph)
    eng = Engram(client, graph, HashEmbedder())
    g = eng.create_scope("global", "global")
    a = eng.create_scope("project", "a", inherits=[g])
    b = eng.create_scope("project", "b", inherits=[g])
    c = eng.create_scope("project", "c", inherits=[a, b, g])  # diamond
    other = eng.create_scope("project", "other", inherits=[g])
    rng = random.Random(3)
    scopes = [g, a, b, c, other]
    now = 100
    for i in range(70):  # a hub: 70 facts, distinct predicates, spread over scopes
        eng.assert_fact("Mega Hub", f"attr {i}", f"hub value {i} {rng.random():.4f}", scopes[i % 5], "a1", now=now)
    for j in range(25):  # mid and small entities
        for k in range(1 + j % 6):
            eng.assert_fact(f"Thing {j}", f"attr {k}", f"thing {j} value {k} {rng.random():.4f}", scopes[(j + k) % 5], "a1", now=now)
    for i in range(6):  # superseded history
        eng.assert_fact("Thing 3", "attr 0", f"newer value {i}", c, "a1", now=now + 1 + i, supersede=True)
    texts = ["Mega Hub attr 7", "Thing 3 attr 0", "hub value", "Thing 12 value", "attr 2", "zzz unrelated"]
    emb = HashEmbedder()
    checked = diffs = 0
    for text in texts:
        q = emb.embed([text])[0]
        for scope in (c, a, g):
            for allv in (False, True):
                for damp in (True, False):
                    for statuses, t in ((["active"], 200), (["active", "superseded", "dormant"], 101)):
                        for lim, sibs in ((8, 5), (60, 5), (120, 3), (5, 1)):
                            p = {"q": list(q), "scope_id": scope, "all": allv, "over": 64, "lim": lim, "t": t, "statuses": statuses,
                                 "boost": 0.05, "sibs": sibs, "kw_ids": [], "damp": damp, "smax": 3.912}
                            ro = client.graph(graph).ro_query(old_lib.get("engram/recall").statements[0], p).result_set
                            rn = client.graph(graph).ro_query(new_lib.get("engram/recall").statements[0], p).result_set
                            hdr = ["fact_id", "entity_id", "entity_name", "predicate", "value", "status", "confidence", "valid_from", "valid_to", "scope_id", "score", "rank", "via", "raw"]

                            def norm(rs):
                                out = {}
                                for r in rs:
                                    d = r[:14]
                                    out[d[0]] = (round(d[10], 6), round(d[11], 6), d[12] is None, d[13], round(r[-2], 6) if isinstance(r[-2], float) else r[-2], round(r[-3], 6) if isinstance(r[-3], float) else r[-3], tuple(sorted(r[-1])))
                                return out

                            no, nn = norm(ro), norm(rn)
                            checked += 1
                            if no.keys() != nn.keys() or any(no[k][:4] != nn[k][:4] or no[k][4:6] != nn[k][4:6] or no[k][6] != nn[k][6] for k in no):
                                diffs += 1
                                if diffs <= 5:
                                    print("DIFF", text, scope, allv, damp, statuses, lim, sibs, len(no), len(nn))
                                    for k in sorted(set(no) | set(nn)):
                                        if no.get(k) != nn.get(k):
                                            print("   ", k, no.get(k), nn.get(k))
                                            break
    print(f"checked {checked} parameter sets, differing {diffs}")
finally:
    client.delete_graph(graph)
