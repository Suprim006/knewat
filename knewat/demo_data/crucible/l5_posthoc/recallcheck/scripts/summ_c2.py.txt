import glob
import json

for p in sorted(glob.glob("C:/Change_folder_after_HDD/wt-recall/knewat/demo_data/crucible/l6/*-c*/results.json")):
    d = json.load(open(p))
    name = p.replace(chr(92), "/").split("/")[-2]
    if "C" not in d["families"]:
        continue
    arms = d["families"]["C"]["arms"]
    print(name, d["config"].get("embedder"))
    for a, v in arms.items():
        bd = v["by_depth"]
        print(
            f"   {a:12s} strict {[bd[x]['strict'] for x in '123']} hit5 {[bd[x]['hit5'] for x in '123']}"
            f" hit_any {[bd[x]['hit_any'] for x in '123']} to {v['all']['timeouts']}"
            f" lat_p50 {[bd[x]['latency_ms']['p50'] for x in '123']} stmts {v['all']['statements']['mean']}"
        )
