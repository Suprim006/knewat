"""summ_l5.py <glob-substring>...: one line per (run, mode)."""

import glob
import json
import sys

root = "C:/Change_folder_after_HDD/wt-recall/knewat/demo_data/crucible/l5"
extra = sys.argv[1:]
paths = []
for pat in extra:
    paths += sorted(glob.glob(pat if "/" in pat or "\\" in pat else f"{root}/*{pat}*/results.json"))
for p in paths:
    d = json.load(open(p))
    name = p.replace("\\", "/").split("/")[-2]
    if "forgetting" in d and d["forgetting"]:
        f = d["forgetting"]
        c = f["config"]
        for mode, m in f["modes"].items():
            r = m["recall"]
            print(
                f"{name:42s} lw={c.get('lexical_weight', 'n/a')} {mode:11s} r@1={r['all']['recall@1']*100:.1f} r@5={r['all']['recall@5']*100:.1f}"
                f" noise={(m.get('noise_share') or 0)*100:.1f} crit_loss={m.get('critical_probe_loss')} same/other={m.get('above_same_entity')}/{m.get('above_other_entity')}"
            )
    for key in ("hub",):
        if key in d and d[key]:
            h = d[key]
            for arm, m in h.get("arms", h.get("results", {})).items():
                print(f"{name:42s} hub {arm}: " + f"r@1={m['recall@1']*100:.1f} r@5={m['recall@5']*100:.1f} hub_above={m['hub_fact_above_target']*100:.1f}")
